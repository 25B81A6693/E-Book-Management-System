    <script>

        // ==================================
        // INITIAL EBOOK DATA
        // ==================================

        let books =
            JSON.parse(
                localStorage.getItem("ebooks")
            ) || [

                {
                    id: 1,

                    title:
                        "Introduction to HTML",

                    author:
                        "John Smith",

                    category:
                        "Programming",

                    progress: 80
                },


                {
                    id: 2,

                    title:
                        "JavaScript Basics",

                    author:
                        "David Miller",

                    category:
                        "Programming",

                    progress: 45
                },


                {
                    id: 3,

                    title:
                        "The Art of Photography",

                    author:
                        "Sarah Wilson",

                    category:
                        "Photography",

                    progress: 100
                },


                {
                    id: 4,

                    title:
                        "Business Management",

                    author:
                        "Robert Brown",

                    category:
                        "Business",

                    progress: 20
                }

            ];


        let editId = null;



        // ==================================
        // SAVE DATA
        // ==================================

        function saveData() {

            localStorage.setItem(
                "ebooks",
                JSON.stringify(books)
            );

        }



        // ==================================
        // DISPLAY BOOKS
        // ==================================

        function displayBooks() {

            const grid =
                document.getElementById(
                    "ebookGrid"
                );


            const search =
                document
                    .getElementById(
                        "searchInput"
                    )
                    .value
                    .toLowerCase();


            const category =
                document
                    .getElementById(
                        "categoryFilter"
                    )
                    .value;


            const filteredBooks =
                books.filter(book => {

                    const titleMatch =
                        book.title
                            .toLowerCase()
                            .includes(search);


                    const authorMatch =
                        book.author
                            .toLowerCase()
                            .includes(search);


                    const categoryMatch =
                        category === "all" ||
                        book.category === category;


                    return (
                        (titleMatch ||
                        authorMatch) &&
                        categoryMatch
                    );

                });



            grid.innerHTML = "";



            if (
                filteredBooks.length === 0
            ) {

                grid.innerHTML = `

                    <div class="empty">

                        <h2>
                            📚 No Ebooks Found
                        </h2>

                        <p>
                            Try another search.
                        </p>

                    </div>

                `;

                updateStatistics();

                return;
            }



            filteredBooks.forEach(
                (book, index) => {

                    const card =
                        document.createElement(
                            "div"
                        );


                    card.className =
                        "ebook-card";


                    const coverClass =
                        "cover" +
                        ((index % 4) + 1);



                    card.innerHTML = `

                        <div
                            class="book-cover ${coverClass}">

                            ${book.title}

                        </div>


                        <div
                            class="ebook-content">

                            <h3>
                                ${book.title}
                            </h3>


                            <div class="author">

                                By ${book.author}

                            </div>


                            <span class="category">

                                ${book.category}

                            </span>


                            <div
                                class="progress-info">

                                <span>
                                    Progress
                                </span>

                                <strong>
                                    ${book.progress}%
                                </strong>

                            </div>


                            <div
                                class="progress-container">

                                <div
                                    class="progress"
                                    style="
                                        width:
                                        ${book.progress}%
                                    ">

                                </div>

                            </div>


                            <div
                                class="card-buttons">


                                <button
                                    class="read-btn"
                                    onclick="
                                        readBook(
                                            ${book.id}
                                        )
                                    ">

                                    📖 Read

                                </button>


                                <button
                                    class="edit-btn"
                                    onclick="
                                        editBook(
                                            ${book.id}
                                        )
                                    ">

                                    ✏️

                                </button>


                                <button
                                    class="delete-btn"
                                    onclick="
                                        deleteBook(
                                            ${book.id}
                                        )
                                    ">

                                    🗑️

                                </button>


                            </div>

                        </div>

                    `;


                    grid.appendChild(card);

                }
            );


            updateStatistics();

        }



        // ==================================
        // UPDATE STATISTICS
        // ==================================

        function updateStatistics() {


            // Total books

            document.getElementById(
                "totalBooks"
            ).textContent =
                books.length;



            // Categories

            const categories =
                new Set(
                    books.map(
                        book => book.category
                    )
                );


            document.getElementById(
                "totalCategories"
            ).textContent =
                categories.size;



            // Completed

            const completed =
                books.filter(
                    book =>
                        book.progress === 100
                ).length;


            document.getElementById(
                "completedBooks"
            ).textContent =
                completed;



            // Reading

            const reading =
                books.filter(
                    book =>
                        book.progress > 0 &&
                        book.progress < 100
                ).length;


            document.getElementById(
                "readingBooks"
            ).textContent =
                reading;



            updateCategories();

        }



        // ==================================
        // CATEGORY FILTER
        // ==================================

        function updateCategories() {

            const select =
                document.getElementById(
                    "categoryFilter"
                );


            const oldValue =
                select.value;


            const categories =
                [
                    ...new Set(
                        books.map(
                            book =>
                                book.category
                        )
                    )
                ];


            select.innerHTML = `

                <option value="all">
                    All Categories
                </option>

            `;


            categories.forEach(
                category => {

                    const option =
                        document.createElement(
                            "option"
                        );


                    option.value =
                        category;


                    option.textContent =
                        category;


                    select.appendChild(
                        option
                    );

                }
            );


            if (
                categories.includes(
                    oldValue
                )
            ) {

                select.value =
                    oldValue;

            } else {

                select.value =
                    "all";

            }

        }



        // ==================================
        // OPEN MODAL
        // ==================================

        function openModal() {

            editId = null;


            document.getElementById(
                "modalTitle"
            ).textContent =
                "Add Ebook";


            document.getElementById(
                "ebookForm"
            ).reset();


            document.getElementById(
                "bookProgress"
            ).value = 0;


            document.getElementById(
                "modal"
            ).style.display =
                "flex";

        }



        // ==================================
        // CLOSE MODAL
        // ==================================

        function closeModal() {

            document.getElementById(
                "modal"
            ).style.display =
                "none";

        }



        // ==================================
        // ADD / UPDATE BOOK
        // ==================================

        document
            .getElementById("ebookForm")
            .addEventListener(
                "submit",
                function(event) {

                    event.preventDefault();


                    const title =
                        document
                            .getElementById(
                                "bookTitle"
                            )
                            .value
                            .trim();


                    const author =
                        document
                            .getElementById(
                                "bookAuthor"
                            )
                            .value
                            .trim();


                    const category =
                        document
                            .getElementById(
                                "bookCategory"
                            )
                            .value
                            .trim();


                    let progress =
                        parseInt(
                            document
                                .getElementById(
                                    "bookProgress"
                                )
                                .value
                        ) || 0;


                    progress =
                        Math.max(
                            0,
                            Math.min(
                                100,
                                progress
                            )
                        );



                    // EDIT

                    if (editId !== null) {

                        const book =
                            books.find(
                                book =>
                                    book.id ===
                                    editId
                            );


                        book.title =
                            title;

                        book.author =
                            author;

                        book.category =
                            category;

                        book.progress =
                            progress;


                    }

                    // ADD

                    else {

                        books.push({

                            id:
                                Date.now(),

                            title:
                                title,

                            author:
                                author,

                            category:
                                category,

                            progress:
                                progress

                        });

                    }



                    saveData();

                    closeModal();

                    displayBooks();

                }
            );



        // ==================================
        // EDIT BOOK
        // ==================================

        function editBook(id) {

            const book =
                books.find(
                    book =>
                        book.id === id
                );


            if (!book) return;


            editId = id;


            document.getElementById(
                "modalTitle"
            ).textContent =
                "Edit Ebook";


            document.getElementById(
                "bookTitle"
            ).value =
                book.title;


            document.getElementById(
                "bookAuthor"
            ).value =
                book.author;


            document.getElementById(
                "bookCategory"
            ).value =
                book.category;


            document.getElementById(
                "bookProgress"
            ).value =
                book.progress;


            document.getElementById(
                "modal"
            ).style.display =
                "flex";

        }



        // ==================================
        // DELETE BOOK
        // ==================================

        function deleteBook(id) {

            const book =
                books.find(
                    book =>
                        book.id === id
                );


            if (!book) return;


            const answer =
                confirm(
                    `Delete "${book.title}"?`
                );


            if (!answer) return;


            books =
                books.filter(
                    book =>
                        book.id !== id
                );


            saveData();

            displayBooks();

        }



        // ==================================
        // READ BOOK
        // ==================================

        function readBook(id) {

            const book =
                books.find(
                    book =>
                        book.id === id
                );


            if (!book) return;


            if (
                book.progress >= 100
            ) {

                alert(
                    "This ebook is already completed!"
                );

                return;
            }


            book.progress =
                Math.min(
                    100,
                    book.progress + 10
                );


            saveData();

            displayBooks();


            alert(
